/* Minebuns Mobile GUI Patch
   원본 Minebuns 스크립트의 맨 마지막 })(); 바로 뒤에 붙이세요.
   원본 모듈/기능은 삭제하지 않고 모바일 메뉴만 추가합니다.
*/
(() => {
  "use strict";

  const start = () => {
    const gui = window.minebuns?.moduleManager?.modules?.ClickGUI;
    if (!gui) return false;
    if (document.getElementById("minebuns-mobile-menu")) return true;

    const style = document.createElement("style");
    style.id = "minebuns-mobile-style";
    style.textContent = `
      #minebuns-mobile-menu {
        position:fixed;top:12px;right:12px;z-index:2147483000;
        border:1px solid rgba(129,225,255,.8);border-radius:12px;
        padding:11px 16px;color:#fff;background:rgba(12,18,26,.96);
        font:700 14px system-ui,sans-serif;box-shadow:0 4px 18px #0008;
        touch-action:manipulation;-webkit-tap-highlight-color:transparent
      }
      #minebuns-mobile-tabs {
        position:fixed;z-index:2147482999;left:8px;right:8px;top:58px;
        display:none;gap:5px;padding:7px;border-radius:12px;
        background:rgba(10,14,20,.97);overflow-x:auto;
        -webkit-overflow-scrolling:touch
      }
      #minebuns-mobile-tabs button {
        flex:1 0 auto;min-width:54px;border:1px solid #42515e;
        border-radius:8px;padding:10px 8px;color:#fff;background:#202b36;
        font:600 12px system-ui,sans-serif;touch-action:manipulation
      }
      #minebuns-mobile-tabs button.active {
        background:linear-gradient(90deg,#40beff,#81e1ff);
        color:#06121b;border-color:transparent
      }
      @media(max-width:700px),(pointer:coarse) {
        .gui-panel {
          position:fixed!important;left:8px!important;right:8px!important;
          top:108px!important;width:auto!important;max-width:none!important;
          max-height:calc(100dvh - 120px)!important;overflow-y:auto!important;
          border:1px solid rgba(129,225,255,.35);border-radius:12px!important;
          overscroll-behavior:contain
        }
        .gui-header{position:sticky;top:0;z-index:2}
        .gui-button{min-height:43px!important;font-size:16px!important}
      }`;
    document.head.appendChild(style);

    const menu = document.createElement("button");
    menu.id = "minebuns-mobile-menu";
    menu.type = "button";
    menu.textContent = "☰ MENU";

    const tabs = document.createElement("div");
    tabs.id = "minebuns-mobile-tabs";
    const categories = [
      ["Combat","전투"],["Movement","이동"],["Visual","시각"],
      ["Misc","기타"],["Menus","메뉴"]
    ];
    let selected = "Combat";
    let opened = false;

    const getPanels = () => Array.isArray(gui.panels) ? gui.panels : [];
    const getPanelElement = p => p.panel || p.element || p.container ||
      (p.header?.parentElement?.classList?.contains("gui-panel") ? p.header.parentElement : null);

    const showCategory = category => {
      selected = category;
      getPanels().forEach(p => {
        const el = getPanelElement(p);
        if (el) el.style.setProperty("display", opened && p.header?.textContent?.trim() === category ? "block" : "none", "important");
      });
      tabs.querySelectorAll("button").forEach(b => b.classList.toggle("active", b.dataset.category === category));
    };

    categories.forEach(([category,label]) => {
      const b = document.createElement("button");
      b.type = "button"; b.textContent = label; b.dataset.category = category;
      b.addEventListener("click", () => showCategory(category));
      tabs.appendChild(b);
    });

    const setOpen = value => {
      opened = value;
      if (opened) {
        if (!gui.isEnabled) gui.enable();
        tabs.style.display = "flex";
        menu.textContent = "✕ 닫기";
        requestAnimationFrame(() => showCategory(selected));
      } else {
        tabs.style.display = "none";
        menu.textContent = "☰ MENU";
        getPanels().forEach(p => {
          const el = getPanelElement(p);
          if (el) el.style.setProperty("display","none","important");
        });
        if (gui.isEnabled) {
          const oldDisable = gui.onDisable;
          gui.onDisable = function () { return oldDisable.call(this, false); };
          try { gui.disable(); } finally { gui.onDisable = oldDisable; }
        }
      }
    };

    menu.addEventListener("click", () => setOpen(!opened));
    document.body.appendChild(menu);
    document.body.appendChild(tabs);

    // ClickGUI가 단축키 등으로 열리거나 닫힐 때 메뉴 상태 동기화
    const sync = () => {
      const enabled = !!gui.isEnabled;
      if (enabled && !opened) {
        opened = true; tabs.style.display = "flex"; menu.textContent = "✕ 닫기";
        showCategory(selected);
      } else if (!enabled && opened) {
        opened = false; tabs.style.display = "none"; menu.textContent = "☰ MENU";
      }
    };
    setInterval(sync, 350);
    window.addEventListener("resize", () => { if (opened) showCategory(selected); });
    return true;
  };

  let tries = 0;
  const timer = setInterval(() => {
    tries++;
    if (start() || tries >= 100) clearInterval(timer);
  }, 250);
})();
